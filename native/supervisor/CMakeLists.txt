cmake_minimum_required(VERSION 3.21)
project(pi_job_supervisor LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows SDK/toolchain required; portable contract tests are separate")
endif()
add_executable(pi-job-supervisor main.cpp)
target_compile_features(pi-job-supervisor PRIVATE cxx_std_17)
target_compile_definitions(pi-job-supervisor PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(pi-job-supervisor PRIVATE advapi32)
if(MSVC)
  target_compile_options(pi-job-supervisor PRIVATE /W4 /permissive- /sdl /guard:cf)
  target_link_options(pi-job-supervisor PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf /MANIFESTUAC:level='asInvoker')
endif()
